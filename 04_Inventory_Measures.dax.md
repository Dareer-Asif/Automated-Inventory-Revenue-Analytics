**///MEASURE 1****Total Quantity Sold:=**SUM(Sales\_Transactions\[Quantity])**///MEASURE 2****Total Quantity Received:=**SUM(Purchase\_Orders\[Quantity\_Received])**///MEASURE 3****Current Stock Level:=**\[Total Quantity Received]-\[Total Quantity Sold]**///MEASURE 4****Stock Status:=**VAR SelectedMinStock= MAX(Dim\_Product\[Min\_Stock\_Level])ReturnIF(ISBLANK(\[Current Stock Level]),"No Transaction Data",IF(\[Current Stock Level]<=SelectedMinStock,"⚠️ Reorder Required","✅ Sufficient Stock"))**///MEASURE 5****Inventory Valuation:=**SUMX(&#x20;   Dim\_Product,&#x20;   \[Current Stock Level] \* CALCULATE(AVERAGE(Purchase\_Orders\[Unit\_Cost])))**///MEASURE 6****Total Sales Revenue:=**SUMX(Sales\_Transactions,Sales\_Transactions\[Quantity]\*\[Unit\_Price])**///MEASURE 7****Total COGS:=**SUMX(Sales\_Transactions, Sales\_Transactions\[Quantity]\*RELATED(Avg\_Unit\_Cost\[Avg Unit Cost]))**///MEASURE 8****Gross Profit:=**\[Total Sales Revenue]-\[Total COGS]**///MEASURE 9****Gross Profit %:=**DIVIDE(\[Gross Profit],\[Total Sales Revenue],0)**///MEASURE 10****Inventory Turnover Ratio:=**DIVIDE(\[Total COGS], \[Inventory Valuation],0)
